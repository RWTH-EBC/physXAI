1""" 

2Construction of recurrent (multi step) models. 

3 

4Two keras models are built: 

5 

6- The **step model** `rnn_step` maps `[sequence, *states] -> [prediction, *states]` for a sequence of 

7 any length. It is the model that is saved and used in AgentLib-MPC, which applies it one step at a 

8 time and warms its states up on the measured past. 

9- The **sequence model** wraps the step model for training and evaluation. It starts from zero 

10 states, warms up on the first `warmup` steps of a sequence (not part of the loss) and predicts the 

11 following `horizon` steps: 

12 

13 - An output 'Change(T)' is accumulated to the trajectory of T, starting from the measured value 

14 T(t0), so the model can be trained on the trajectory. 

15 - If a predicted feature is also an input of the model, it is fed back during the prediction 

16 (closed loop), as in an MPC. During the warmup the measured values are used. 

17"""

69def build_step_model( 

70 config: dict, td, feature_architecture: list[Expr] | None = None, name: str = STEP_MODEL_NAME 

71) -> keras.Model: 

72 """ 

73 Builds the step model `[sequence, *states] -> [prediction, *states]`. 

74 

75 Args: 

76 config: Settings, validated by `RNNModelConstruction_config` (rnn_units, rnn_layer). 

77 td: `TrainingDataMultiStep`, used for the normalization of in- and outputs. 

78 feature_architecture: Expressions (`physXAI.expr`) building the inputs of the recurrent layer 

79 out of the input features. Defaults to the normalized inputs. 

80 name: Name of the model. 

81 """ 

82 config = RNNModelConstruction_config.model_validate(config).model_dump() 

83 units, rnn_layer = config["rnn_units"], config["rnn_layer"] 

84 n_inputs, n_outputs = len(td.inputs), len(td.outputs) 

85 X = td.X_train_single 

86 targets = td.targets("train").reshape(-1, n_outputs) 

87 

88 # AgentLib-MPC recognizes input layers by 'input' in their name 

89 sequence = keras.Input(shape=(None, n_inputs), name="sequence_input") 

90 states = [keras.Input(shape=(units,), name=f"state_input_{i}") for i in range(number_of_states(rnn_layer))] 

91 

92 if feature_architecture is None: 

93 normalization = keras.layers.Normalization(mean=X.mean(axis=0), variance=np.maximum(X.var(axis=0), 1e-12)) 

94 features = normalization(sequence) 

95 else: 

96 # the expressions slice the features they need out of the last axis, so they work on 

97 # sequences as well as on single time steps 

98 ctx = BuildContext(sequence, td.inputs, {c: X[:, i] for i, c in enumerate(td.inputs)}) 

99 channels = [ctx.tensor(expression) for expression in feature_architecture] 

100 features = channels[0] if len(channels) == 1 else keras.layers.Concatenate()(channels) 

101 

102 rnn = RNN_LAYERS[rnn_layer](units, return_sequences=True, return_state=True) 

103 hidden, *new_states = rnn(features, initial_state=states) 

104 prediction = keras.layers.Dense(n_outputs)(hidden) 

105 # plain floats: a Rescaling layer holding tensors cannot be serialized 

106 scale = [float(v) for v in np.maximum(targets.std(axis=0), 1e-12)] 

107 offset = [float(v) for v in targets.mean(axis=0)] 

108 prediction = keras.layers.Rescaling( 

109 scale=scale if n_outputs > 1 else scale[0], offset=offset if n_outputs > 1 else offset[0] 

110 )(prediction) 

111 return keras.Model([sequence, *states], [prediction, *new_states], name=name)

137def build_pcnn_step_model(config: dict, td, terms: list[PCNNTerm], name: str = STEP_MODEL_NAME) -> keras.Model: 

138 """ 

139 Builds the step model of a physically consistent neural network (PCNN, Di Natale et al. 2022, 

140 Applied Energy 325, 119806), `[sequence, E, *states of f] -> [change of T, E, *states of f]`. 

141 

142 The predicted feature T (input and output `Change(T)`) is the sum of the unforced dynamics D and the 

143 energy accumulator E: 

144 

145 E(k+1) = E(k) + sum_heating a * u(k) - sum_cooling d * u(k) + sum_losses b * (T_x(k) - T(k)) 

146 D(k+1) = D(k) + f(D(k), x(k)), D(k) = T(k) - E(k) 

147 

148 f is a recurrent network on D and the remaining inputs x, which excludes the controls and the 

149 temperatures of the losses. With a, b, d > 0 and 1 - sum b > 0, T increases with heating, decreases 

150 with cooling and increases with the temperatures T_x over the whole horizon, for any f. The parameters 

151 work in physical units, f on normalized values. 

152 

153 Only keras layers translated by AgentLib-MPC are used (input slices, constants, subtract, multiply, 

154 concatenate, dense, simple_rnn, normalization, rescaling, add), the layer names carry their types. 

155 

156 Args: 

157 config: Settings, validated by `RNNModelConstruction_config` (rnn_units, rnn_layer of f). 

158 td: `TrainingDataMultiStep` with one output `Change(T)` and T as input. 

159 terms: The terms of the energy accumulator. 

160 name: Name of the model. 

161 """ 

162 from physXAI.models.ann.keras_models.keras_models import ConstantLayer, InputSliceLayer 

163 

164 config = RNNModelConstruction_config.model_validate(config).model_dump() 

165 units, rnn_layer = config["rnn_units"], config["rnn_layer"] 

166 n_inputs = len(td.inputs) 

167 X = td.X_train_single 

168 feature = td.inputs.index(td.features[0]) 

169 change = td.targets("train").reshape(-1) 

170 

171 sequence = keras.Input(shape=(None, n_inputs), name="sequence_input") 

172 energy_state = keras.Input(shape=(1,), name="state_input_0") 

173 states = [keras.Input(shape=(units,), name=f"state_input_{i + 1}") for i in range(number_of_states(rnn_layer))] 

174 

175 # --- physics-inspired module: increment of the energy accumulator in physical units ----------- 

176 temperature = InputSliceLayer([feature], name="input_slice_pcnn_feature")(sequence) 

177 scaled = list() 

178 for j, term in enumerate(terms): 

179 column = InputSliceLayer([term.input], name=f"input_slice_pcnn_term_{j}")(sequence) 

180 if term.kind == "loss": 

181 column = keras.layers.Subtract(name=f"subtract_pcnn_gradient_{j}")([column, temperature]) 

182 initial = keras.layers.Multiply(name=f"multiply_pcnn_scale_{j}") 

183 constant = ConstantLayer(value=PCNN_TERMS[term.kind] * term.initial, shape=(1,), name=f"constant_pcnn_scale_{j}") 

184 scaled.append(initial([column, constant(column)])) 

185 physics = scaled[0] if len(scaled) == 1 else keras.layers.Concatenate(name="concatenate_pcnn_physics")(scaled) 

186 increment = keras.layers.Dense( 

187 1, use_bias=False, kernel_initializer="ones", kernel_constraint=keras.constraints.NonNeg(), 

188 name=PCNN_PHYSICS_LAYER, 

189 )(physics) 

190 # E(k+1) = E(k) + increment(k): a linear recurrent layer with fixed weights 1 

191 accumulator = keras.layers.SimpleRNN( 

192 1, activation="linear", use_bias=False, kernel_initializer="ones", recurrent_initializer="ones", 

193 return_sequences=True, return_state=True, trainable=False, name="simple_rnn_pcnn_energy", 

194 ) 

195 energy_next, new_energy_state = accumulator(increment, initial_state=[energy_state]) 

196 energy = keras.layers.Subtract(name="subtract_pcnn_energy")([energy_next, increment]) 

197 unforced = keras.layers.Subtract(name="subtract_pcnn_unforced")([temperature, energy]) 

198 

199 # --- black-box module f(D, x) ---------------------------------------------------------------- 

200 used = {feature, *(term.input for term in terms)} 

201 others = [i for i in range(n_inputs) if i not in used] 

202 columns = [feature, *others] 

203 if others: 

204 disturbances = InputSliceLayer(others, name="input_slice_pcnn_disturbances")(sequence) 

205 features = keras.layers.Concatenate(name="concatenate_pcnn_unforced")([unforced, disturbances]) 

206 else: 

207 features = unforced 

208 normalized = keras.layers.Normalization( 

209 mean=X[:, columns].mean(axis=0), variance=np.maximum(X[:, columns].var(axis=0), 1e-12), 

210 name="normalization_pcnn", 

211 )(features) 

212 prefix = {"RNN": "simple_rnn", "GRU": "gru", "LSTM": "lstm"}[rnn_layer] 

213 rnn = RNN_LAYERS[rnn_layer](units, return_sequences=True, return_state=True, name=f"{prefix}_pcnn_unforced") 

214 hidden, *new_states = rnn(normalized, initial_state=states) 

215 unforced_change = keras.layers.Dense(1, name="dense_pcnn_unforced")(hidden) 

216 unforced_change = keras.layers.Rescaling( 

217 scale=float(max(change.std(), 1e-12)), offset=0.0, name="rescaling_pcnn" 

218 )(unforced_change) 

219 prediction = keras.layers.Add(name="add_pcnn_change")([unforced_change, increment]) 

220 return keras.Model([sequence, energy_state, *states], [prediction, new_energy_state, *new_states], name=name)

265@keras.saving.register_keras_serializable(package="physXAI") 

266class Accumulate(keras.layers.Layer): 

267 """Trajectory of the features: start value plus the accumulated changes (absolute outputs as they are).""" 

268 

269 def __init__(self, difference: list[bool], **kwargs): 

270 super().__init__(**kwargs) 

271 self.difference = [bool(d) for d in difference] 

272 

273 def call(self, prediction, initial): 

274 mask = keras.ops.convert_to_tensor(np.asarray(self.difference, dtype="float32").reshape(1, 1, -1)) 

275 accumulated = initial[:, None, :] + keras.ops.cumsum(prediction, axis=1) 

276 return mask * accumulated + (1.0 - mask) * prediction 

277 

278 def get_config(self): 

279 return {**super().get_config(), "difference": self.difference}

282class ClosedLoopCell(keras.layers.Layer): 

283 """ 

284 RNN cell applying the step model one step at a time and feeding predicted features back. 

285 

286 State: `[values of the predicted features, *states of the step model]`. 

287 Input: one row of input features; the fed back features are replaced by the predicted values. 

288 Output: [trajectory of the predicted features, outputs of the step model]. 

289 """ 

290 

291 def __init__(self, step_model: keras.Model, spec: SequenceSpec, **kwargs): 

292 super().__init__(**kwargs) 

293 self.step_model = step_model 

294 slots = np.zeros((spec.n_outputs, spec.n_inputs), dtype="float32") 

295 for i, j in spec.feedback: 

296 slots[j, i] = 1.0 

297 self.slots = slots 

298 self.mask = slots.sum(axis=0, keepdims=True) 

299 self.difference = np.asarray(spec.difference, dtype="float32").reshape(1, -1) 

300 self.state_size = [spec.n_outputs, *state_sizes(step_model)] 

301 self.output_size = 2 * spec.n_outputs 

302 

303 def call(self, inputs, states): 

304 values, *rnn_states = states 

305 row = inputs * (1.0 - self.mask) + keras.ops.matmul(values, self.slots) 

306 prediction, *rnn_states = self.step_model([row[:, None, :], *rnn_states]) 

307 prediction = prediction[:, 0, :] 

308 values = self.difference * (values + prediction) + (1.0 - self.difference) * prediction 

309 return keras.ops.concatenate([values, prediction], axis=-1), [values, *rnn_states]

332def build_sequence_model( 

333 step_model: keras.Model, spec: SequenceSpec, warmup: int, horizon: int, trajectory: bool = True 

334) -> keras.Model: 

335 """ 

336 Wraps the step model for the prediction of sequences (see the module docstring). 

337 

338 Args: 

339 step_model: The step model. 

340 spec: In- and outputs of the model. 

341 warmup: Number of warmup steps. 

342 horizon: Number of predicted steps. 

343 trajectory: If True (and the start values are known), the model returns the trajectories of 

344 the predicted features, else the outputs of the step model (e.g. the changes). 

345 

346 Returns: 

347 keras.Model: [sequence (batch, warmup + horizon, inputs), start values (batch, outputs) if 

348 `needs_initial`] -> (batch, horizon, outputs). 

349 """ 

350 trajectory = trajectory and spec.absolute 

351 sequence = keras.Input(shape=(warmup + horizon, spec.n_inputs), name="sequence") 

352 inputs = [sequence] 

353 initial = None 

354 if needs_initial(spec, trajectory): 

355 initial = keras.Input(shape=(spec.n_outputs,), name="initial") 

356 inputs.append(initial) 

357 states = ZeroStates(state_sizes(step_model))(sequence) 

358 

359 if not spec.feedback: 

360 # open loop: the whole sequence in one pass, starting from zero states 

361 prediction, *_ = step_model([sequence, *states]) 

362 if warmup: 362 ↛ 364line 362 didn't jump to line 364, because the condition on line 362 was never false

363 prediction = TimeSlice(warmup)(prediction) 

364 if trajectory and any(spec.difference): 

365 prediction = Accumulate(spec.difference)(prediction, initial) 

366 output = prediction 

367 else: 

368 # warmup on the measured past, then closed loop over the horizon 

369 if warmup: 

370 _, *states = step_model([TimeSlice(0, warmup)(sequence), *states]) 

371 future = TimeSlice(warmup)(sequence) if warmup else sequence 

372 cell = ClosedLoopCell(step_model, spec) 

373 result = keras.layers.RNN(cell, return_sequences=True)(future, initial_state=[initial, *states]) 

374 start = 0 if trajectory else spec.n_outputs 

375 output = TakeFeatures(start, start + spec.n_outputs)(result) 

376 return keras.Model(inputs, output, name="rnn_sequence")

379def find_step_model(model: keras.Model) -> keras.Model: 

380 """The step model of a sequence model (or the model itself, if it is a step model).""" 

381 if model.name == STEP_MODEL_NAME: 

382 return model 

383 try: 

384 return model.get_layer(STEP_MODEL_NAME) 

385 except ValueError: 

386 pass 

387 for layer in model.layers: 387 ↛ 390line 387 didn't jump to line 390, because the loop on line 387 didn't complete

388 if isinstance(layer, keras.layers.RNN) and isinstance(layer.cell, ClosedLoopCell): 

389 return layer.cell.step_model 

390 raise ValueError(f"The model '{model.name}' holds no step model '{STEP_MODEL_NAME}'.")

393def sequence_inputs(td, split: str, spec: SequenceSpec, trajectory: bool, horizon=None, stride=None): 

394 """Inputs of the sequence model and the matching truth for a split (None if the split is empty).""" 

395 data = td.sequences(split, horizon, stride) 

396 if data is None: 

397 return None 

398 X, targets, initial = data 

399 trajectory = trajectory and spec.absolute 

400 x = [X, initial] if needs_initial(spec, trajectory) else X 

401 y = td.trajectory(targets, initial) if trajectory else targets 

402 return x, y

405def predict_sequences(model: keras.Model, td, horizon: int, stride: int) -> dict: 

406 """ 

407 Predicts the trajectories of all sequences of `horizon` steps. 

408 

409 Returns: 

410 dict: {split: (prediction, truth)}, both of shape (sequences, horizon, outputs). 

411 """ 

412 from physXAI.preprocessing.splitting import SPLITS 

413 

414 step_model = find_step_model(model) 

415 spec = SequenceSpec.from_training_data(td) 

416 network = build_sequence_model(step_model, spec, td.warmup_width, horizon, trajectory=True) 

417 predictions = dict() 

418 for split in SPLITS: 

419 data = sequence_inputs(td, split, spec, True, horizon, stride) 

420 if data is not None: 

421 predictions[split] = (network.predict(data[0], verbose=0, batch_size=1024), data[1]) 

422 return predictions

425def evaluate_sequences(model: keras.Model, td) -> None: 

426 """ 

427 Evaluates a recurrent model on the sequences of `td` as it is applied in an MPC: warmup on the 

428 measured past, closed loop for fed back features, accumulated changes. Stores the predictions, 

429 the metrics (`MetricsMultiStep`) and the errors over the evaluation horizon (`HorizonMetrics`). 

430 """ 

431 from physXAI.evaluation.metrics import MetricsMultiStep 

432 from physXAI.evaluation.rollout import HorizonMetrics 

433 

434 label = predict_sequences(model, td, td.label_width, td.stride) 

435 

436 def prediction(split): 

437 return label[split][0] if split in label else None 

438 

439 td.add_predictions(prediction("train"), prediction("val"), prediction("test")) 

440 td.add_metrics(MetricsMultiStep(td)) 

441 

442 if td.eval_horizon == td.label_width and td.eval_stride == td.stride: 

443 horizon = label 

444 else: 

445 horizon = predict_sequences(model, td, td.eval_horizon, td.eval_stride) 

446 td.horizon_predictions = horizon 

447 if horizon: 447 ↛ exitline 447 didn't return from function 'evaluate_sequences', because the condition on line 447 was never false

448 features = td.features if td.absolute else td.outputs 

449 td.horizon_metrics = HorizonMetrics(horizon, td.time_step, features)
